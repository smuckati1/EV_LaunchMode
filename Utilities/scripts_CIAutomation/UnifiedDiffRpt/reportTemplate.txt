    <script>
    function navigateToWebview(frame,path) {
        id = (frame == 2) ? "mine" : "theirs";
        const webview = document.getElementById(id);
        const webviewWindow = webview.contentWindow;            
        webviewWindow.slwebview.unhighlightAll("slwebview:" + path);
        webviewWindow.slwebview.open("slwebview:" + path);
    }
    function highlightToWebview(frame,path) {
        id = (frame == 2) ? "mine" : "theirs";
        color = (frame == 2) ? "cyan" : "yellow";
        const webview = document.getElementById(id);
        const webviewWindow = webview.contentWindow;
        webviewWindow.slwebview.unhighlightAll("slwebview:" + path);
        webviewWindow.slwebview.highlight("slwebview:" + path,color);
    }
